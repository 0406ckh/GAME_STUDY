# 상하좌우 애니메이션 실습

<br>

```C#
// Player.cs
using Unity.VisualScripting;
using UnityEngine;

// 캐릭터가 어느 방향으로 가는지 관리 해주는 것도 필요함
public enum Dir
{
    Up,
    Down,
    Left,
    Right
}

// 방향에 추가하여 상태까지 관리
public enum State
{
    Idle,
    Move,
    Skill
}

public class Player : MonoBehaviour
{
    Animator _animator;
    SpriteRenderer _spriteRenderer;

    Dir _dir;
    State _state;

    void Start()
    {
        _animator = GetComponent<Animator>();
        _spriteRenderer = GetComponent<SpriteRenderer>();
        // Play는 애니메이션을 실행해줌
        // String으로 애니메이션 이름을 찾아줌
        // String으로 찾으면 조금 느림
        // _animator.Play("SideMove");

        // 아래처럼 하면 정수로 변환 후 맞춰서 플레이
        // 미세하지만 String으로 비교하는 것보다 효율적임
        // int stringToMoveHash = Animator.StringToHash("SideMove");
        // _animator.Play(stringToMoveHash);
    }

    void Update()
    {
        UpdateInput();
        UpdateAnimation();
    }

    // input도 한번에 관리하는 것이 좋음
    void UpdateInput()
    {
        if (Input.GetKey(KeyCode.W))
        {
            // _animator.Play("UpMove");
            _dir = Dir.Up;
            _state = State.Move;
        }
        else if (Input.GetKey(KeyCode.S))
        {
            // _animator.Play("DownMove");
            _dir = Dir.Down;
            _state = State.Move;
        }
        else if (Input.GetKey(KeyCode.A))
        {
            // 왼쪽 오른쪽은 같은 애니메이션임
            // 따라서 Sprite Renderer에서 Flip을 사용하여 반대 방향으로 바꿔줘야함
            // _animator.Play("SideMove");
            // _spriteRenderer.flipX = true;
            _dir = Dir.Left;
            _state = State.Move;
        }
        else if (Input.GetKey(KeyCode.D))
        {
            // 왼쪽 오른쪽은 같은 애니메이션임
            // 따라서 Sprite Renderer에서 Flip을 사용하여 반대 방향으로 바꿔줘야함
            // _animator.Play("SideMove");
            // _spriteRenderer.flipX = false;
            _dir = Dir.Right;
            _state = State.Move;
        }
        else
        {
            _state = State.Idle;
        }
    }

    // 방향 정보를 애니메이션 코드와 같이 써주는 것은 좋은 습관이 아님
    // 한번에 애니메이션을 업데이트 해서 관리하는 것이 버그 확률이 떨어짐
    void UpdateAnimation()
    {
        if (_state == State.Idle)
        {
            switch(_dir)
            {
                case Dir.Up:
                    _animator.Play("UpIdle");
                    break;
                case Dir.Down:
                    _animator.Play("DownIdle");
                    break;
                case Dir.Left:
                    _animator.Play("SideIdle");
                    _spriteRenderer.flipX = true;
                    break;
                case Dir.Right:
                    _animator.Play("SideIdle");
                    _spriteRenderer.flipX = false;
                    break;
            }
        }
        else if (_state == State.Move)
        {
            switch (_dir)
            {
                case Dir.Up:
                    _animator.Play("UpMove");
                    break;
                case Dir.Down:
                    _animator.Play("DownMove");
                    break;
                case Dir.Left:
                    _animator.Play("SideMove");
                    _spriteRenderer.flipX = true;
                    break;
                case Dir.Right:
                    _animator.Play("SideMove");
                    _spriteRenderer.flipX = false;
                    break;
            }
        }
        else if ( _state == State.Skill)
        {

        }
    }
}
```
