# 코루틴
실행하는 함수의 중간 상태를 저장할 수 있음

<br>

### 예제 코드
```C#
// MainScript.cs
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.InputSystem;
public enum Step
{
    Step1_Countdown, // 스텝 1
    Step2_Wait,      // 스텝 2
    Step3_WaitSpace, // 스텝 3
}
public class MainScript : MonoBehaviour
{
    // 만들어볼 콘텐츠
    // 스텝 1. 5 4 3 2 1 (1초마다 카운트)
    // 스텝 2. 3초 정도 대기
    // 스텝 3. Space 눌렀을 때만 진행

    Step _step = Step.Step1_Countdown;

    void Start()
    {
        
    }

    // 1초마다 카운트
    int _counter = 5;
    float _sumTime = 1;  // 경과 시간

    void Update()
    {
        // 스텝 1일때만 실행
        if(_step == Step.Step1_Countdown)
        {
            // 1초가 지나면 경과 시간을 초기화 하고 카운터 출력
            _sumTime += Time.deltaTime;
            if (_sumTime >= 1)
            {
                Debug.Log(_counter);
                _counter--;
                _sumTime = 0;

                // 카운터가 0이되면 스텝 2로 넘어가기
                if (_counter == 0)
                {
                    _step = Step.Step2_Wait;
                }
            }
        }
        // 스텝 2일 때 실행
        else if (_step == Step.Step2_Wait)
        {
            _sumTime += Time.deltaTime;
            // 3초 대기
            if(_sumTime >=3)
            {
                _step = Step.Step3_WaitSpace;

            }
        }
        // 스텝 3일 때 실행
        else if(_step == Step.Step3_WaitSpace)
        {
            // 스페이스바를 누르고 있을 때
            if (Input.GetKeyDown(KeyCode.Space))
            {
                Debug.Log("Congratulations!");
            }
        }
    }
}
```
**위와같이 코드를 짜면 생기는 문제점**  
* 코드가 복잡하고 어지러움  
* 상태를 별도로 관리해주어야 함  
* 따로 관리해 줘야 하는 변수들이 많음(`_counter`, `_sumTime` 등)  

<br>

### 코루틴 함수
```C#
// MainScript.cs
using System.Collections.Generic;
using System.Collections;
using UnityEngine;
using UnityEngine.InputSystem;
using Unity.VisualScripting;

public class MainScript : MonoBehaviour
{
    // 만들어볼 콘텐츠
    // 스텝 1. 5 4 3 2 1 (1초마다 카운트)
    // 스텝 2. 3초 정도 대기
    // 스텝 3. Space 눌렀을 때만 진행

    // 멤버변수로 코루틴 저장
    Coroutine _coTutorial;

    void Start()
    {
        // 코루틴 함수를 호출해야함
        // StartCoroutine의 리턴값은 Coroutine
        _coTutorial = StartCoroutine(CoTutorial());
        
        // 코루틴을 종료하고 싶으면 멤버변수로 코루틴을 저장하고 아래 코드로 강제종료 가능
        // StopCoroutine(_coTutorial);
    }

    // 코루틴은 반드시 리턴값이 IEnumerator여야 함
    // 함수 앞에 Co를 붙여주는 것이 컨벤션
    IEnumerator CoTutorial()
    {
        // 스텝 1
        for(int count = 5; count > 0; count--)
        {
            Debug.Log(count);
            // new WaitForSeconds(1): 1초동안 기다렸다 돌아오라는 조건
            yield return new WaitForSeconds(1);
        }

        // 스텝 2
        Debug.Log("Waiting 3 seconds");
        yield return new WaitForSeconds(3);
        Debug.Log("Waiting 3 seconds Ended");

        // 스텝 3
        while (true)
        {
            // 스페이스바가 눌리면 while 탈출
            if (Input.GetKeyDown(KeyCode.Space))
            {
                break;
            }

            yield return null;
        }
        // 위의 스텝 3 코드는 아래와 같이 변경할 수 있음
        // yield return new WaitUntil(() => { return Input.GetKeyDown(KeyCode.Space); });

        Debug.Log("Congratulations!");

        // 코루틴 리턴은 yield return;
        // using System.Collections 추가 필요함
        // 뒤에 조건 추가 가능
        // null은 한 프레임 뒤에 다시 실행
        yield return null;
    }


    void Update()
    {
        
    }
}
```
예제 코드와 똑같은 실행  
임시 데이터를 줄이고 코드를 간편하게 만들 수 있어 가독성도 올라감  
업데이트마다 매번 시간을 더하는 것보다 효율이나 성능이 좋음
