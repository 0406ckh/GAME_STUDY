# 공격 애니메이션 실습

<br>

### 공격 모션 추가
```C#
// Player.cs
using Unity.VisualScripting;
using UnityEngine;

public enum Dir
{
    Up,
    Down,
    Left,
    Right
}

public enum State
{
    Idle,
    Move,
    Skill
}

public class Player : MonoBehaviour
{
    Animator _animator;
    SpriteRenderer _spriteRenderer;

    Dir _dir;
    State _state;

    void Start()
    {
        _animator = GetComponent<Animator>();
        _spriteRenderer = GetComponent<SpriteRenderer>();
    }

    void Update()
    {
        UpdateInput();
        UpdateAnimation();
    }

    void UpdateInput()
    {
        if (Input.GetKey(KeyCode.W))
        {
            _dir = Dir.Up;
            _state = State.Move;
        }
        else if (Input.GetKey(KeyCode.S))
        {
            _dir = Dir.Down;
            _state = State.Move;
        }
        else if (Input.GetKey(KeyCode.A))
        {
            _dir = Dir.Left;
            _state = State.Move;
        }
        else if (Input.GetKey(KeyCode.D))
        {
            _dir = Dir.Right;
            _state = State.Move;
        }
        // 공격 모션 추가
        else if (Input.GetKeyDown(KeyCode.Space))
        {
            _state = State.Skill;
        }
        else
        {
            _state = State.Idle;
        }
    }

    void UpdateAnimation()
    {
        if (_state == State.Idle)
        {
            switch(_dir)
            {
                case Dir.Up:
                    _animator.Play("UpIdle");
                    break;
                case Dir.Down:
                    _animator.Play("DownIdle");
                    break;
                case Dir.Left:
                    _animator.Play("SideIdle");
                    _spriteRenderer.flipX = true;
                    break;
                case Dir.Right:
                    _animator.Play("SideIdle");
                    _spriteRenderer.flipX = false;
                    break;
            }
        }
        else if (_state == State.Move)
        {
            switch (_dir)
            {
                case Dir.Up:
                    _animator.Play("UpMove");
                    break;
                case Dir.Down:
                    _animator.Play("DownMove");
                    break;
                case Dir.Left:
                    _animator.Play("SideMove");
                    _spriteRenderer.flipX = true;
                    break;
                case Dir.Right:
                    _animator.Play("SideMove");
                    _spriteRenderer.flipX = false;
                    break;
            }
        }
        // 공격 모션 추가
        else if ( _state == State.Skill)
        {
            switch (_dir)
            {
                case Dir.Up:
                    _animator.Play("UpAttack");
                    break;
                case Dir.Down:
                    _animator.Play("DownAttack");
                    break;
                case Dir.Left:
                    _animator.Play("SideAttack");
                    _spriteRenderer.flipX = true;
                    break;
                case Dir.Right:
                    _animator.Play("SideAttack");
                    _spriteRenderer.flipX = false;
                    break;
            }
        }
    }
}
```
실행해보면 스페이스바를 눌러도 공격이 안됨  
키보드를 떼는 순간 다른 애니메이션으로 덮이기 때문  

<br>

### 해결
`UpdateInput()`에 `if (_state == State.Skill) return;`을 넣어줘서 idle 상태로 돌아가지 않게 설정하면 무한하게 공격하는 것을 볼 수 있음  
그 후 애니메이션 파일을 클릭해서 `Loop Time`을 꺼주면 공격 모션이 한번 실행되고 idle로 돌아가지 않고 멈춰버림  

<br>

**해결방법1**  
애니메이션 파일 위에 `Length`라고 애니메이션이 실행되는 시간이 있음  
코드상에서 시간만큼 공격 모션을 취하게 하고 그 후 Idle 상태로 돌아가게 하면 해결할 수 있음  
하지만 이렇게 하면 관리가 어려워짐  

<br>

**해결방법2**  
`Animation` 창에서 마지막 모션의 시간 밑을 우클릭 하고 `Add Animation Event`  
`Function`에 공격이 끝났다는 것을 알려주는 함수를 넣어줌(지금은 `OnAttackEnded`)  
`Player.cs`에 `OnAttackEnded` 함수 추가  
```C#
// Player.cs 완성 코드
using Unity.VisualScripting;
using UnityEngine;

public enum Dir
{
    Up,
    Down,
    Left,
    Right
}

public enum State
{
    Idle,
    Move,
    Skill
}

public class Player : MonoBehaviour
{
    Animator _animator;
    SpriteRenderer _spriteRenderer;

    Dir _dir;
    State _state;

    void Start()
    {
        _animator = GetComponent<Animator>();
        _spriteRenderer = GetComponent<SpriteRenderer>();
    }

    void Update()
    {
        UpdateInput();
        UpdateAnimation();
    }

    void UpdateInput()
    {
        if (Input.GetKey(KeyCode.W))
        {
            _dir = Dir.Up;
            _state = State.Move;
        }
        else if (Input.GetKey(KeyCode.S))
        {
            _dir = Dir.Down;
            _state = State.Move;
        }
        else if (Input.GetKey(KeyCode.A))
        {
            _dir = Dir.Left;
            _state = State.Move;
        }
        else if (Input.GetKey(KeyCode.D))
        {
            _dir = Dir.Right;
            _state = State.Move;
        }
        // 공격 모션 추가
        else if (Input.GetKeyDown(KeyCode.Space))
        {
            _state = State.Skill;
        }
        else
        {
            if (_state == State.Skill) return;

            _state = State.Idle;
        }
    }

    void UpdateAnimation()
    {
        if (_state == State.Idle)
        {
            switch(_dir)
            {
                case Dir.Up:
                    _animator.Play("UpIdle");
                    break;
                case Dir.Down:
                    _animator.Play("DownIdle");
                    break;
                case Dir.Left:
                    _animator.Play("SideIdle");
                    _spriteRenderer.flipX = true;
                    break;
                case Dir.Right:
                    _animator.Play("SideIdle");
                    _spriteRenderer.flipX = false;
                    break;
            }
        }
        else if (_state == State.Move)
        {
            switch (_dir)
            {
                case Dir.Up:
                    _animator.Play("UpMove");
                    break;
                case Dir.Down:
                    _animator.Play("DownMove");
                    break;
                case Dir.Left:
                    _animator.Play("SideMove");
                    _spriteRenderer.flipX = true;
                    break;
                case Dir.Right:
                    _animator.Play("SideMove");
                    _spriteRenderer.flipX = false;
                    break;
            }
        }
        // 공격 모션 추가
        else if ( _state == State.Skill)
        {
            switch (_dir)
            {
                case Dir.Up:
                    _animator.Play("UpAttack");
                    break;
                case Dir.Down:
                    _animator.Play("DownAttack");
                    break;
                case Dir.Left:
                    _animator.Play("SideAttack");
                    _spriteRenderer.flipX = true;
                    break;
                case Dir.Right:
                    _animator.Play("SideAttack");
                    _spriteRenderer.flipX = false;
                    break;
            }
        }
    }

    void OnAttackEnded()
    {
        _state = State.Move;
    }
}
```
무한 공격을 하지 않고 일회 공격 후 멈추지 않고 다시 Idle 상태로 돌아감
